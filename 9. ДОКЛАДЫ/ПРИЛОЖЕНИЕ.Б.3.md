Вот полная программа на C++:

```cpp
#include <iostream>
#include <iomanip>
#include <cmath>
#include <map>
#include <string>
#include <cstdio>

// ============================================================
// МОДЕЛЬ ВСЕЛЕННОЙ Б — ВЫЧИСЛЕНИЕ ТОЧЕК В ОБА НАПРАВЛЕНИЯ
// Версия 3.1-fix — C++
// Исправление: отрицательный остаток от деления (% 2)
// ============================================================

struct Row {
    double a, b, c;
};

// --- ИСХОДНЫЕ ДАННЫЕ (Зерно) ---
std::map<int, Row> seed = {
    {1, {-3.0,   -0.5,   -2.5}},
    {2, {-1.667, -0.833,  6.25}},
    {3, {-2.64,  -7.91,   1.842}},
};

const int N_FORWARD  = 30;
const int N_BACKWARD = 15;
const double LAM0    = 215.4;

// --- БЕЗОПАСНАЯ ПРОВЕРКА НЕЧЁТНОСТИ ---
// В C++ остаток от деления отрицательных чисел может быть
// отрицательным: (-3) % 2 == -1, а не 1.
// Поэтому используем формулу: ((x % 2) + 2) % 2
inline bool is_odd(int x) {
    return ((x % 2) + 2) % 2 == 1;
}

// --- 1D ПРАВИЛО (колонка c) ---
// Вперёд:  нечётный шаг → c^2,  чётный шаг → c^{1/3}
// Назад:   обратный квадрат → -sqrt(|c|),  обратный корень → c^3

double c_forward(double c, int n) {
    if (is_odd(n)) {
        return c * c;
    } else {
        if (c >= 0) return std::pow(c, 1.0 / 3.0);
        return -std::pow(-c, 1.0 / 3.0);
    }
}

double c_backward(double c, int n) {
    // n = 0: обр.корень, n = -1: обр.квадрат, n = -2: обр.корень, ...
    if (is_odd(n)) {
        return -std::sqrt(std::abs(c));   // обратный «квадрат»
    } else {
        return c * c * c;                 // обратный «корень»
    }
}

// --- 2D ПРАВИЛО (колонки a, b — первичные) ---
// Режим «квадрат» (нечётный):  S = a - b;  a' = 2S/3;  b' = S/3
// Режим «корень»  (чётный):    S = |a+b|^{2/3};  a' = S/3;  b' = S

Row ab_forward(double a, double b, int n) {
    if (is_odd(n)) {
        double S = a - b;
        return {2.0 * S / 3.0, S / 3.0, 0};
    } else {
        double S = std::pow(std::abs(a + b), 2.0 / 3.0);
        return {S / 3.0, S, 0};
    }
}

Row ab_backward(double a, double b, int n) {
    if (is_odd(n)) {
        // Обратный «квадрат»: b' = S_prev, b_prev = S/3 → S_prev = 3*b'
        // S_prev = |3*b|^1.5, a_prev = 2*S_prev/3
        double S_prev = std::pow(std::abs(3.0 * b), 1.5);
        return {2.0 * S_prev / 3.0, S_prev, 0};
    } else {
        // Обратный «корень»: b' = S, a' = S/3
        // S_prev = |b'|^1.5, a_prev = 2*S_prev/3, b_prev = S_prev/3
        double S_prev = std::pow(std::abs(b), 1.5);
        return {2.0 * S_prev / 3.0, S_prev / 3.0, 0};
    }
}

// --- ФОРМАТИРОВАНИЕ ---
// Для очень малых и очень больших значений используем научную нотацию

std::string fmt(double v) {
    if (std::isinf(v) || std::isnan(v)) return "  ---  ";

    double av = std::abs(v);
    if (av > 1e15) {
        char buf[32];
        snprintf(buf, sizeof(buf), "%.3e", v);
        return buf;
    }
    if (av > 0 && av < 1e-4) {
        char buf[32];
        snprintf(buf, sizeof(buf), "%.3e", v);
        return buf;
    }
    char buf[32];
    snprintf(buf, sizeof(buf), "%.6f", v);
    return buf;
}

std::string fmt_au(double v) {
    if (std::isinf(v) || std::isnan(v)) return "  ---  ";
    if (v > 1e6) {
        char buf[32];
        snprintf(buf, sizeof(buf), "%.4e", v);
        return buf;
    }
    if (v > 0 && v < 1e-4) {
        char buf[32];
        snprintf(buf, sizeof(buf), "%.4e", v);
        return buf;
    }
    char buf[32];
    snprintf(buf, sizeof(buf), "%.4f", v);
    return buf;
}

int main() {
    std::map<int, Row> data;

    // Заполнение исходными данными
    for (auto& [n, row] : seed) {
        data[n] = row;
    }

    // --- ВПЕРЁД от N=3 ---
    double a = data[3].a, b = data[3].b, c = data[3].c;
    for (int n = 4; n < 4 + N_FORWARD; ++n) {
        Row r = ab_forward(a, b, n - 1);
        a = r.a; b = r.b;
        c = c_forward(c, n - 1);
        data[n] = {a, b, c};
    }

    // --- НАЗАД от N=1 ---
    a = data[1].a; b = data[1].b; c = data[1].c;
    for (int n = 0; n > -N_BACKWARD - 1; --n) {
        Row r = ab_backward(a, b, n + 1);
        a = r.a; b = r.b;
        c = c_backward(c, n + 1);
        data[n] = {a, b, c};
    }

    // --- ВЫВОД ---
    std::cout << "=" << std::string(100, '=') << "\n";
    std::cout << std::setw(4) << "N" << " | "
              << std::setw(18) << "a" << " | "
              << std::setw(18) << "b" << " | "
              << std::setw(20) << "c" << " | "
              << std::setw(14) << "|c| AU" << " | "
              << std::setw(7) << "λ°" << "\n";
    std::cout << "=" << std::string(100, '=') << "\n";

    // Назад (от -N_BACKWARD до -1)
    for (int n = -N_BACKWARD; n < 0; ++n) {
        if (data.find(n) == data.end()) continue;
        Row& row = data[n];
        double lam = std::fmod(LAM0 + (n - 1) * 30.0, 360.0);
        if (lam < 0) lam += 360.0;
        std::cout << std::setw(4) << n << " | "
                  << std::setw(18) << fmt(row.a) << " | "
                  << std::setw(18) << fmt(row.b) << " | "
                  << std::setw(20) << fmt(row.c) << " | "
                  << std::setw(14) << fmt_au(std::abs(row.c)) << " | "
                  << std::setw(7) << std::fixed << std::setprecision(1) << lam << "\n";
    }

    std::cout << "-" << std::string(100, '-') << "\n";

    // Вперёд (от 1 до 3+N_FORWARD)
    for (int n = 1; n < 4 + N_FORWARD; ++n) {
        if (data.find(n) == data.end()) continue;
        Row& row = data[n];
        double lam = std::fmod(LAM0 + (n - 1) * 30.0, 360.0);
        if (lam < 0) lam += 360.0;
        std::string star = (n <= 3) ? " ★" : "";
        std::cout << std::setw(4) << n << " | "
                  << std::setw(18) << fmt(row.a) << " | "
                  << std::setw(18) << fmt(row.b) << " | "
                  << std::setw(20) << fmt(row.c) << " | "
                  << std::setw(14) << fmt_au(std::abs(row.c)) << " | "
                  << std::setw(7) << std::fixed << std::setprecision(1) << lam
                  << star << "\n";
    }

    std::cout << "=" << std::string(100, '=') << "\n";
    std::cout << "★ — исходные строки (Зерно)\n";
    std::cout << "Малые значения отображаются в научной нотации (e-notation)\n";

    // --- ДЕТАЛЬНАЯ ПРОВЕРКА ШАГОВ НАЗАД ---
    std::cout << "\n=== ДЕТАЛЬНАЯ ПРОВЕРКА ШАГОВ НАЗАД ===\n";
    for (int n = 0; n > -8; --n) {
        if (data.find(n) == data.end()) continue;
        Row& row = data[n];
        int step = n + 1;
        std::string step_type = is_odd(step) ? "обр.квадрат" : "обр.корень";
        std::cout << "  N=" << std::setw(3) << n
                  << ": шаг=" << std::setw(12) << step_type
                  << ", a=" << fmt(row.a)
                  << ", b=" << fmt(row.b)
                  << ", c=" << fmt(row.c) << "\n";
    }

    return 0;
}
```

Компиляция и запуск:

```bash
g++ -std=c++17 -O2 -o universe_b universe_b.cpp
./universe_b
```

Программа выведет таблицу из ~48 строк: 15 шагов назад (N = −15…−1), три исходные (N = 1–3, помечены ★) и 30 шагов вперёд (N = 4…33). Колонки — `a`, `b`, `c`, `|c|` в AU и эклиптическая долгота `λ°`. Для значений свыше `10^15` автоматически переключается на экспоненциальную запись, чтобы корректно обрабатывать точки в межзвёздном пространстве при движении назад.
